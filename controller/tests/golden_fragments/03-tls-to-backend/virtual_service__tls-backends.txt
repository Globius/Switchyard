frontend vs_tls-backends
    mode http
    bind 192.0.2.22:443 ssl crt-list /etc/switchyard/certs/tls-backends.crtlist alpn h2,http/1.1,acme-tls/1
    option httplog
    log-format "%ci:%cp [%tr] %f %b/%s %TR/%Tw/%Tc/%Tr/%Ta %ST %B request_id=%ID sni=%{+Q}[ssl_fc_sni] host=%{+Q}[var(txn.host)] upstream=%[bc_dst] upstream_port=%[bc_dst_port] retries=%rc %tsc %ac/%fc/%bc/%sc %sq/%bq %hr %hs %{+Q}r"
    option forwardfor
    http-request del-header X-Forwarded-For
    http-request set-header X-Forwarded-Proto https if { ssl_fc }
    http-request set-header X-Forwarded-Proto http if !{ ssl_fc }
    http-request set-var(txn.host) req.hdr(host),field(1,:),lower
    use_backend %[var(txn.host),map(/etc/switchyard/haproxy/maps/tls-backends-host.map)] if { var(txn.host),map(/etc/switchyard/haproxy/maps/tls-backends-host.map) -m found }
