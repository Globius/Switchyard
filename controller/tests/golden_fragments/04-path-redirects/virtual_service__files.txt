frontend vs_files
    mode http
    bind 192.0.2.23:80
    bind 192.0.2.23:443 ssl crt-list /etc/switchyard/certs/files.crtlist alpn http/1.1,acme-tls/1
    option httplog
    log-format "%ci:%cp [%tr] %f %b/%s %TR/%Tw/%Tc/%Tr/%Ta %ST %B request_id=%ID sni=%{+Q}[ssl_fc_sni] host=%{+Q}[var(txn.host)] upstream=%[bc_dst] upstream_port=%[bc_dst_port] retries=%rc %tsc %ac/%fc/%bc/%sc %sq/%bq %hr %hs %{+Q}r"
    option forwardfor
    http-request return status 200 content-type text/plain lf-string "%[path,map(/etc/switchyard/haproxy/maps/acme-http01.map)]" if { dst_port 80 } { path,map(/etc/switchyard/haproxy/maps/acme-http01.map) -m found }
    http-request del-header X-Forwarded-For
    http-request set-header X-Forwarded-Proto https if { ssl_fc }
    http-request set-header X-Forwarded-Proto http if !{ ssl_fc }
    http-request set-var(txn.host) req.hdr(host),field(1,:),lower
    http-request redirect prefix "https://%[var(txn.host)]" code 301 if { dst_port 80 }
    http-request redirect location /remote.php/dav/ code 301 if { path_beg /.well-known/carddav }
    http-request redirect location /remote.php/dav/ code 301 if { path_beg /.well-known/caldav }
    http-request redirect location https://files.example.test/new code 302 if { var(txn.host) -m str files.example.test } { path /old }
    use_backend pool_docs if { var(txn.host) -m str files.example.test } { path_beg /docs }
    use_backend %[var(txn.host),map(/etc/switchyard/haproxy/maps/files-host.map)] if { var(txn.host),map(/etc/switchyard/haproxy/maps/files-host.map) -m found }
